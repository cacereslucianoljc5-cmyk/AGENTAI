------------------------------------------------------------------------
 0) IDEA GENERAL (cómo funciona)
------------------------------------------------------------------------
- Es una web con 2 pestañas: CHAT (LLM) e IMÁGENES (texto -> imagen).
- El navegador llama a 2 rutas internas:
     /api/chat   -> reenvía el chat a Pollinations (texto)
     /api/image  -> devuelve una URL de imagen de Pollinations (sin logo)
- Los LÍMITES diarios se controlan en 2 sitios:
     * Servidor (lib/ratelimit.ts): tope por IP, en memoria (backstop).
     * Cliente  (app/page.tsx): contador visible guardado en localStorage.
- Los LOGOS se sirven por CDN (jsDelivr) desde el repo público, así se ven
  en vivo aunque el deploy no esté conectado a Git. (Ver sección 6.)
- No hay variables de entorno ni secretos. Se despliega tal cual en Vercel.

        cp -r AGENTAI proyecto-2   (o descarga el repo y renómbralo)

   3. Crea un repo NUEVO en GitHub para proyecto-2 y súbelo:
        cd proyecto-2
        git init
        git add -A
        git commit -m "init proyecto-2"
        git branch -M main
        git remote add origin https://github.com/TU_USUARIO/proyecto-2.git
        git push -u origin main
   4. Importa proyecto-2 en Vercel como un proyecto Vercel NUEVO
      (otra URL). HoodAgentAi sigue intacto.
   5. IMPORTANTE con los logos por CDN: la constante CDN en app/page.tsx
      apunta al repo AGENTAI. Para proyecto-2 tienes 2 opciones:
        a) Poner tus propias imágenes en public/ y cambiar el <img>
           para usar rutas locales: src="/mascot.png"  (recomendado).
        b) Actualizar la constante CDN para que apunte a TU nuevo repo:
           https://cdn.jsdelivr.net/gh/TU_USUARIO/proyecto-2@main/public


------------------------------------------------------------------------
 6) CÓMO FUNCIONAN LOS LOGOS POR CDN (jsDelivr)
------------------------------------------------------------------------
 - Las imágenes viven en public/ del repo PÚBLICO.
 - Se sirven por: https://cdn.jsdelivr.net/gh/USUARIO/REPO@COMMIT/public/archivo
 - Se usa un COMMIT fijo (SHA) para que no dependa de caché de rama.
 - Ventaja: se ven en vivo aunque el deploy NO esté conectado a Git.
 - Si conectas el repo a Vercel (deploy desde Git), puedes cambiar las
   URLs del CDN por rutas locales: src="/mascot.png".
 - Si actualizas una imagen: haz commit, copia el nuevo SHA
   (git rev-parse HEAD) y reemplázalo en la constante CDN de page.tsx.

######## lib/ratelimit.ts ########
// Límite diario por IP en memoria (backstop simple para tier gratis).
// Para límites estrictos y persistentes: Upstash Redis o Neon Postgres.
type Bucket = { count: number; day: string };
const store = new Map<string, Bucket>();

export const LIMITS = {
  chat: 40,
  image: 20,
} as const;

export type Feature = keyof typeof LIMITS;

function today(): string {
  return new Date().toISOString().slice(0, 10);
}

export function getClientIp(req: Request): string {
  const xff = req.headers.get("x-forwarded-for");
  if (xff) return xff.split(",")[0].trim();
  return req.headers.get("x-real-ip") || "anonymous";
}

export type RateResult = { allowed: boolean; remaining: number; limit: number };

export function checkAndConsume(ip: string, feature: Feature): RateResult {
  const limit = LIMITS[feature];
  const key = `${feature}:${ip}`;
  const day = today();
  const bucket = store.get(key);
  if (!bucket || bucket.day !== day) {
    store.set(key, { count: 1, day });
    return { allowed: true, remaining: limit - 1, limit };
  }
  if (bucket.count >= limit) return { allowed: false, remaining: 0, limit };
  bucket.count += 1;
  return { allowed: true, remaining: limit - bucket.count, limit };
}

######## app/api/chat/route.ts ########
import { NextResponse } from "next/server";
import { checkAndConsume, getClientIp, LIMITS } from "@/lib/ratelimit";

export const runtime = "nodejs";
export const maxDuration = 30;

const SYSTEM_PROMPT =
  "Eres HoodAgent, un asistente de IA rápido, directo y útil. Respondes en el " +
  "idioma del usuario. Eres claro y conciso, con un toque amigable. Nunca " +
  "reveles el proveedor técnico que te ejecuta; simplemente eres HoodAgent.";

type ChatMessage = { role: "user" | "assistant" | "system"; content: string };

export async function POST(req: Request) {
  const ip = getClientIp(req);
  const rate = checkAndConsume(ip, "chat");
  if (!rate.allowed) {
    return NextResponse.json(
      { error: `Has alcanzado el límite diario de ${LIMITS.chat} mensajes. Vuelve mañana.`, remaining: 0, limit: LIMITS.chat },
      { status: 429 }
    );
  }

  let body: { messages?: ChatMessage[] };
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: "Cuerpo inválido" }, { status: 400 });
  }

  const messages = Array.isArray(body.messages) ? body.messages : [];
  if (messages.length === 0)
    return NextResponse.json({ error: "Faltan mensajes" }, { status: 400 });

  const trimmed = messages.slice(-12);
  const payload = {
    model: "openai",
    messages: [{ role: "system", content: SYSTEM_PROMPT }, ...trimmed],
    private: true,
    referrer: "hoodagentai",
  };

  try {
    const resp = await fetch("https://text.pollinations.ai/openai", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    if (!resp.ok) {
      const text = await resp.text().catch(() => "");
      return NextResponse.json(
        { error: "El motor de IA no respondió. Intenta de nuevo.", detail: text.slice(0, 200) },
        { status: 502 }
      );
    }
    const data = await resp.json().catch(() => null);
    const reply =
      data?.choices?.[0]?.message?.content ??
      (typeof data === "string" ? data : null);
    if (!reply)
      return NextResponse.json({ error: "Respuesta vacía del motor de IA." }, { status: 502 });
    return NextResponse.json({ reply, remaining: rate.remaining, limit: rate.limit });
  } catch {
    return NextResponse.json({ error: "Error de red al contactar el motor de IA." }, { status: 502 });
  }
}

######## app/api/image/route.ts ########
import { NextResponse } from "next/server";
import { checkAndConsume, getClientIp, LIMITS } from "@/lib/ratelimit";

export const runtime = "nodejs";
export const maxDuration = 60;

export async function POST(req: Request) {
  const ip = getClientIp(req);
  const rate = checkAndConsume(ip, "image");
  if (!rate.allowed) {
    return NextResponse.json(
      { error: `Has alcanzado el límite diario de ${LIMITS.image} imágenes. Vuelve mañana.`, remaining: 0, limit: LIMITS.image },
      { status: 429 }
    );
  }

  let body: { prompt?: string; ratio?: string };
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: "Cuerpo inválido" }, { status: 400 });
  }

  const prompt = (body.prompt || "").trim();
  if (!prompt) return NextResponse.json({ error: "Escribe una descripción" }, { status: 400 });
  if (prompt.length > 800) return NextResponse.json({ error: "La descripción es demasiado larga" }, { status: 400 });

  const sizes: Record<string, [number, number]> = {
    square: [1024, 1024],
    landscape: [1280, 768],
    portrait: [768, 1280],
  };
  const [width, height] = sizes[body.ratio || "square"] || sizes.square;
  const seed = (Date.now() % 1_000_000) + prompt.length;

  const params = new URLSearchParams({
    width: String(width),
    height: String(height),
    seed: String(seed),
    nologo: "true",
    model: "flux",
    referrer: "hoodagentai",
  });

  const url = `https://image.pollinations.ai/prompt/${encodeURIComponent(prompt)}?${params.toString()}`;
  return NextResponse.json({ url, remaining: rate.remaining, limit: rate.limit });
}
